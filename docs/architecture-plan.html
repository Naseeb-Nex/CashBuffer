🗄️ Final Consolidated Database Schema (PostgreSQL 15+)

Architecture: Multi-Tenant Household Ledger + Ingress Web Cron
Driver Strategy: psycopg 3 (Binary protocol, parameterized async queries)
Zero SQL Alchemy: Raw SQL with explicit async connection pooling

Below is the finalized schema including Household multi-tenancy, provider-neutral internal users, accounts, categories, deduplicated transactions, and self-contained DB cron tables.

Identity Boundary: Internal UUID `users.id`
Provider Mapping: `users.idp_user_id`
Vendor Lock-in: Avoided
Provider-Neutral Authentication Flow
  1. Frontend sends provider JWT to FastAPI.
  2. Auth adapter validates JWT signature and issuer using configured JWKS.
  3. Adapter extracts provider subject claim into idp_user_id.
  4. Backend executes SELECT id FROM users WHERE idp_user_id = %s; creates the internal user if absent.
  5. All application tables reference internal users.id UUID. No transaction, household, OAuth, or permission table references Kinde directly.
  6. Changing Kinde to another identity provider changes only the auth adapter and issuer configuration; ledger data remains unchanged.

Security rule: never accept user_id or household_id from request body as authority. Derive current internal user from validated JWT, then verify household membership for every query.

Complete SQL DDL Schema (Run directly with psycopg 3) 
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

-- 1. Tenant Root: Household
CREATE TABLE IF NOT EXISTS households (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    base_currency VARCHAR(3) DEFAULT 'USD' NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL
);

-- 2. Internal Auth Identity (provider-neutral)
-- `id` is CashBuffer's internal identity. Never expose or use provider IDs as FKs.
CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    idp_user_id VARCHAR(255) NOT NULL, -- Kinde/Auth0/etc. subject (`sub` claim)
    email VARCHAR(255) UNIQUE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT uq_users_idp_user UNIQUE (idp_user_id)
);
CREATE INDEX IF NOT EXISTS idx_users_idp_user ON users(idp_user_id);

-- 3. Household Membership & Permissions
CREATE TABLE IF NOT EXISTS household_members (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    household_id UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    role VARCHAR(20) DEFAULT 'owner' NOT NULL, -- 'owner', 'admin', 'member', 'viewer'
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT uq_household_user UNIQUE (household_id, user_id)
);
CREATE INDEX IF NOT EXISTS idx_members_user ON household_members(user_id);

-- 4. Accounts (Compound key: [id, household_id] prevents cross-tenant leaks)
CREATE TABLE IF NOT EXISTS accounts (
    id UUID DEFAULT gen_random_uuid(),
    household_id UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    type VARCHAR(30) NOT NULL, -- 'checking', 'savings', 'credit_card', 'cash'
    currency VARCHAR(3) DEFAULT 'INR' NOT NULL,
    current_balance NUMERIC(15, 2) DEFAULT 0.00 NOT NULL,
    is_active BOOLEAN DEFAULT TRUE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    PRIMARY KEY (id, household_id)
);

-- 5. Categories (Scoped to Household + optional hierarchical nesting)
CREATE TABLE IF NOT EXISTS categories (
    id UUID DEFAULT gen_random_uuid(),
    household_id UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
    parent_id UUID,
    name VARCHAR(100) NOT NULL,
    color VARCHAR(20),
    icon VARCHAR(50),
    is_income BOOLEAN DEFAULT FALSE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    PRIMARY KEY (id, household_id),
    CONSTRAINT fk_category_parent FOREIGN KEY (parent_id, household_id)
        REFERENCES categories(id, household_id) ON DELETE SET NULL,
    CONSTRAINT uq_category_name UNIQUE (household_id, name)
);

-- 6. Budgets
CREATE TABLE IF NOT EXISTS budgets (
    id UUID DEFAULT gen_random_uuid(),
    household_id UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    PRIMARY KEY (id, household_id)
);

-- 7. Merchants / Payees
CREATE TABLE IF NOT EXISTS merchants (
    id UUID DEFAULT gen_random_uuid(),
    household_id UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
    name VARCHAR(150) NOT NULL,
    default_category_id UUID,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    PRIMARY KEY (id, household_id),
    CONSTRAINT fk_merchant_cat FOREIGN KEY (default_category_id, household_id)
        REFERENCES categories(id, household_id) ON DELETE SET NULL,
    CONSTRAINT uq_household_merchant UNIQUE (household_id, name)
);

-- 8. Transactions (Core Multi-Tenant Ledger with Deduplication Hash)
-- NOTE: ALL INGRESS TRANSACTIONS LAND DIRECTLY HERE!
CREATE TABLE IF NOT EXISTS transactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    household_id UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
    account_id UUID NOT NULL REFERENCES accounts(id, household_id) ON DELETE CASCADE,
    user_id UUID REFERENCES users(id) ON DELETE SET NULL,         -- linked user who owns ingress
    category_id UUID REFERENCES categories(id, household_id) ON DELETE SET NULL,
    merchant_id UUID REFERENCES merchants(id, household_id) ON DELETE SET NULL,
    amount NUMERIC(15, 2) NOT NULL,                               -- Negative = expense, Positive = income
    date DATE NOT NULL,                                           -- Parsed date from bank email
    notes TEXT,                                                   -- Raw vendor / email snippet
    is_cleared BOOLEAN DEFAULT TRUE NOT NULL,                     -- Bank alert confirmed
    status VARCHAR(50) DEFAULT 'parsed' NOT NULL,                 -- 'parsed', 'needs_review', 'categorized'
    raw_source VARCHAR(50) DEFAULT 'gmail_axis',                 -- Ingress origin
    raw_email_id VARCHAR(100),                                   -- Gmail message ID (e.g. '18dd9c8034b7119e')
    email_url TEXT,                                               -- Direct link: https://mail.google.com/mail/u/0/#all/{raw_email_id}
    dedupe_hash VARCHAR(64) UNIQUE,                              -- SHA-256(household_id + date + amount + vendor)
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT fk_tx_account FOREIGN KEY (account_id, household_id)
        REFERENCES accounts(id, household_id) ON DELETE CASCADE,
    CONSTRAINT fk_tx_category FOREIGN KEY (category_id, household_id)
        REFERENCES categories(id, household_id) ON DELETE SET NULL,
    CONSTRAINT fk_tx_merchant FOREIGN KEY (merchant_id, household_id)
        REFERENCES merchants(id, household_id) ON DELETE SET NULL
);
CREATE INDEX IF NOT EXISTS idx_tx_lookup ON transactions(household_id, date DESC);
CREATE INDEX IF NOT EXISTS idx_tx_account ON transactions(account_id, date DESC);

-- 8b. Quarantined Emails (Landing table ONLY when bank alert email fails parser regex)
CREATE TABLE IF NOT EXISTS quarantined_emails (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    household_id UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
    user_id UUID REFERENCES users(id) ON DELETE SET NULL,
    raw_email_id VARCHAR(100),
    email_url TEXT,                                               -- Direct link: https://mail.google.com/mail/u/0/#all/{raw_email_id}
    email_text TEXT NOT NULL,
    error_reason TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL
);

-- 9. Ingress Configurations (Database-Driven Web Cron)
CREATE TABLE IF NOT EXISTS ingress_configs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    household_id UUID NOT NULL REFERENCES households(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    provider VARCHAR(50) NOT NULL DEFAULT 'gmail',
    is_enabled BOOLEAN NOT NULL DEFAULT TRUE,
    interval_minutes INT NOT NULL DEFAULT 60, -- 15, 30, 60, 360, 1440
    last_run_at TIMESTAMPTZ,
    next_run_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(50) NOT NULL DEFAULT 'idle', -- 'idle', 'running', 'failed'
    last_error TEXT,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT uq_household_ingress UNIQUE (household_id, provider)
);
CREATE INDEX IF NOT EXISTS idx_ingress_due ON ingress_configs(is_enabled, next_run_at) 
    WHERE is_enabled = TRUE;

-- 10. Ingress Sync Checkpoints (Resume without reprocessing)
CREATE TABLE IF NOT EXISTS ingress_sync_state (
    household_id UUID PRIMARY KEY REFERENCES households(id) ON DELETE CASCADE,
    last_history_id VARCHAR(100),
    last_email_internal_date BIGINT,
    total_processed INT DEFAULT 0 NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP NOT NULL
);

📖 End-to-End Data Story: Ingress to Financial Features

Scope: Lifecycle of an Ingressed Transaction
Storage: PostgreSQL (psycopg 3)
Downstream Consumer: Ledger UI + FastMCP AI Tools

1. Ingestion: Bank Email to `transactions` Table

Ingress daemon wakes up on schedule, pulls alerts via Gmail API, and writes directly:

  • Parsing: Extracts amount (debit negative, credit positive), date, raw merchant string.
  • Email Navigation Link: Saves raw_email_id and creates clickable email_url = https://mail.google.com/mail/u/0/#all/{raw_email_id}.
  • Deduplication: Generates dedupe_hash = SHA256(household_id + date + amount + merchant). Repetitive email fetch skips inserting without error.
  • Status: Stored with status = 'parsed'. (If email unparseable, saved into quarantined_emails).
  • Checkpoint: Updates ingress_sync_state.last_email_internal_date.

2. Merchant Matching & Auto-Categorization

Once landed in transactions, the pipeline executes category tagging:

  • Merchant Directory: Looks up merchants table for match (e.g., "Swiggy", "Uber", "Amazon").
  • Direct Link: If known, sets transactions.merchant_id and transactions.category_id = merchants.default_category_id, updates status = 'categorized'.
  • New Merchant Review: If unknown, creates new record in merchants and sets status = 'needs_review'. Shows badge in Web UI for 1-click confirmation.

3. Balance Calculations & Daily Snapshots

Keeps accounts up to date without heavy table-scans:

  • Account Ledger: Updates accounts.current_balance atomically.
  • Daily Snapshots: Aggregates end-of-day balances into account_balances (account_id, household_id, date, balance).
  • Sub-Millisecond Charts: Web dashboard queries account_balances for instantaneous 30-day net worth and burn rate graphs.

4. Budget Enforcement & Household Limits

Uses budgets and budget_categories:

  • Calculates category spending for current calendar cycle against budget_categories.target_amount.
  • Calculates buffer headroom: Remaining Budget = Target - SUM(Debits).
  • Triggers visual warnings in Web UI when category threshold reaches 80% or 100%.

5. AI Advisor via FastMCP Server

Exposes read-only financial tools to external agents (Claude, Pi, Cursor):

  • get_monthly_burn(household_id): Computes net income, expenses, and burn rate.
  • detect_recurring_bills(household_id): Finds recurring bills (rent, Netflix, gym, utilities) by analyzing date intervals.
  • calculate_safe_buffer(household_id): Computes true safe-to-spend cash buffer (Liquid Balances minus Upcoming Bills).
  • Conversational Planning: External agents can answer questions like "Can I buy a new phone this week?" using verified data.

6. User Dashboard Features

Features available to the end user in the Next.js UI:

  • Transaction List: Filter by date/category/merchant, with direct [Open Gmail] ↗ link on every row.
  • Ingress Cron Manager: Toggle automated sync, change interval dropdown (15m, 1h, 24h), or click "Sync Now".
  • Quarantine Inbox: View failed email alerts with error reason, edit and convert into transaction with 1 click.
  • AI Advisor Chat: In-browser conversation assistant to analyze spending and financial goals.

from psycopg_pool import AsyncConnectionPool
from app.infrastructure.config import settings

pool = AsyncConnectionPool(
    conninfo=settings.DATABASE_URL,
    min_size=2,
    max_size=10,
    open=False
)

async def get_db_pool() -> AsyncConnectionPool:
    if pool.closed:
        await pool.open()
    return pool

# apps/ingress/src/infrastructure/persistence/ingress_repository.py
async def fetch_due_ingress_jobs(pool):
    async with pool.connection() as conn:
        async with conn.cursor() as cur:
            await cur.execute("""
                SELECT id, household_id, user_id, interval_minutes
                FROM ingress_configs
                WHERE is_enabled = TRUE AND next_run_at <= CURRENT_TIMESTAMP
                FOR UPDATE SKIP LOCKED;
            """)
            return await cur.fetchall()

Example psycopg 3 Tool Implementation

# apps/mcp/tools/summary_tools.py
from mcp.server.fastmcp import FastMCP
from apps.mcp.db import get_pool

mcp = FastMCP("CashBuffer AI Intel")

@mcp.tool()
async def get_monthly_burn(household_id: str, month: int, year: int) -> dict:
    """Calculates income vs expenses vs remaining buffer for household."""
    pool = await get_pool()
    async with pool.connection() as conn:
        async with conn.cursor() as cur:
            await cur.execute("""
                SELECT 
                    COALESCE(SUM(CASE WHEN amount > 0 THEN amount ELSE 0 END), 0) AS total_income,
                    COALESCE(SUM(CASE WHEN amount < 0 THEN ABS(amount) ELSE 0 END), 0) AS total_expense
                FROM transactions
                WHERE household_id = %s 
                  AND EXTRACT(MONTH FROM date) = %s 
                  AND EXTRACT(YEAR FROM date) = %s;
            """, (household_id, month, year))
            row = await cur.fetchone()
            return {"income": float(row[0]), "expense": float(row[1]), "burn_rate": float(row[1])}
